class Solution {
    public:
    vector<int> kLargest(vector<int>& arr, int k) {
        // code here
        vector<int> ans;
        sort(arr.begin(), arr.end(), greater<int>());
        
        for (int i = 0; i < k; i++) {
            ans.push_back(arr[i]);
        }
        
        return ans;
    }
};
