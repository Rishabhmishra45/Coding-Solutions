class Solution {
  public:
    int minSubarraySum(vector<int> &arr) {
        // code here
        int mini = INT_MAX;
        int prefix = 0;
        for(int i=0; i<arr.size(); i++){
            prefix = prefix + arr[i];
            mini = min(mini,prefix);
            if(prefix > 0){
                prefix = 0;
            }
        }
        return mini;
    }
};
